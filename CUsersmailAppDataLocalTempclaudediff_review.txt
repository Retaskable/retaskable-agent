+/// 1 s, 2 s, 4 s … up to 60 s, each with up to half taken off at random (`random` in 0..1) so agents don't reconnect

+pub fn backoff(attempt: u32, unit: Duration, random: f64) -> Duration {
+    let wait = unit.saturating_mul(2u32.saturating_pow(attempt)).min(unit * 60);
+    wait.mul_f64(1.0 - random.clamp(0.0, 1.0) / 2.0)

+/// `wss://…/api/agents/connect?project=<id>`, no `project` for the agent's own hub.
+fn connect_url(url: &str, project: &str) -> String {
+    let base = format!("{}/api/agents/connect", url.replacen("http", "ws", 1));
+    if project.is_empty() {
+        return base;
+    }
+    match Url::parse_with_params(&base, [("project", project)]) {
+        Ok(u) => u.to_string(),
+        Err(_) => base,
+    }

+/// One project's connection: connects, runs it until it closes, and connects again.
+async fn run(cloud: Arc<Cloud>, project: String, mut opened: Option<watch::Sender<bool>>) {
+    let t = cloud.timing;
+    let log = |msg: String| {
+        let who = if project.is_empty() {
+            String::new()
+        } else {
+            format!("{project}: ")
+        };
+        tracing::info!("{}: {who}{msg}", cloud.label);
+    };
+    let mut attempt = 0u32;
+    let next_backoff = |attempt: &mut u32| {
+        let wait = backoff(*attempt, t.backoff, random());
+        *attempt = attempt.saturating_add(1);
+        wait
+    };
+    let mut stale: Option<String> = None;
+    loop {
+        // A token that never comes (a refresh hanging on the network) is a failed try, not a link that stops.
+        let answer = tokio::time::timeout(t.connect, cloud.tokens.access_token(stale.as_deref()))
+            .await
+            .unwrap_or(TokenAnswer::Unreachable);
+        let wait = match answer {
+            TokenAnswer::Idle => t.idle,
+            TokenAnswer::Unreachable => next_backoff(&mut attempt),
+            TokenAnswer::Token { url, access_token } => {
+                let ws_url = connect_url(&url, &project);
+                // A socket that neither opens nor fails would hold this project off the cloud for good: give it up.
+                match tokio::time::timeout(t.connect, open(&ws_url, &access_token)).await {
+                    Ok(Ok(ws)) => {
+                        log(format!("connected to {url}"));
+                        if let Some(opened) = opened.take() {
+                            let _ = opened.send(true); // the other projects may connect now
+                        }
+                        let since = Instant::now();
+                        let (code, reason) = connected(ws, &cloud.view, &project, t, &log).await;
+                        if since.elapsed() >= t.stable {
+                            attempt = 0;
+                        }
+                        if code == close::TOKEN_EXPIRED {
+                            // The token ran out: the next connect takes a fresh one, at once.
+                            stale = Some(access_token);
+                            Duration::ZERO
+                        } else {
+                            stale = None;
+                            let wait = next_backoff(&mut attempt);
+                            log(format!(
+                                "disconnected: {}; retrying in {} s",
+                                why(code, &reason),
+                                wait.as_secs()
+                            ));
+                            wait
+                        }
+                    }
+                    failed => {
+                        let why = match failed {
+                            Ok(Err(e)) => e,
+                            _ => format!("no answer from {url} in {} s", t.connect.as_secs()),
+                        };
+                        // A connect that never opened may have been refused its token: the next takes a fresh one.
+                        stale = Some(access_token);
+                        let wait = next_backoff(&mut attempt);
+                        log(format!("couldn't connect: {why}; retrying in {} s", wait.as_secs()));
+                        wait
+                    }
+                }
+            }
+        };
+        tokio::time::sleep(wait).await;
+    }
+}

+        _ if reason.is_empty() => format!("closed ({code})"),

+/// Opens the socket, signing in with the access token beside the subprotocol.
+async fn open(ws_url: &str, access_token: &str) -> Result<Socket, String> {
+    let mut request = ws_url.into_client_request().map_err(|e| cause(&e))?;
+    let protocols = HeaderValue::from_str(&format!("{SUBPROTOCOL}, {access_token}")).map_err(|e| cause(&e))?;
+    request.headers_mut().insert("Sec-WebSocket-Protocol", protocols);
+    let (ws, _) = tokio_tungstenite::connect_async(request).await.map_err(|e| cause(&e))?;
+    Ok(ws)
+}

+/// Runs an open connection: hello and a snapshot, then answers what the hub sends, and says it's alive in a quiet
+/// minute (sync-health.md section 3.1). The close code and reason when it closes.
+async fn connected(ws: Socket, view: &AgentView, project: &str, t: Timing, log: &impl Fn(String)) -> (u16, String) {
+    let (mut tx, mut rx) = ws.split();
+    for msg in [view.hello(project), view.snapshot(project)] {
+        if let Err(e) = tx.send(Message::text(msg.to_string())).await {
+            return (1006, cause(&e));
+        }
+    }
+    let mut alive = tokio::time::interval_at(tokio::time::Instant::now() + t.idle, t.idle);
+    loop {
+        let reply = tokio::select! {
+            msg = rx.next() => match msg {
+                Some(Ok(Message::Text(text))) => receive(&text, log),
+                Some(Ok(Message::Close(frame))) => {
+                    return frame.map_or((1005, String::new()), |f| (u16::from(f.code), f.reason.to_string()));
+                }
+                Some(Ok(_)) => None,
+                Some(Err(e)) => return (1006, cause(&e)),
+                None => return (1006, String::new()),
+            },
+            _ = alive.tick() => Some(json!({ "type": "alive" })),
+        };
+        if let Some(reply) = reply
+            && let Err(e) = tx.send(Message::text(reply.to_string())).await
+        {
+            return (1006, cause(&e));
+        }
+    }
+}

+    Some(json!({ "type": "nack", "id": id, "reason": reason }))

+    async fn token(State(seen): State<Arc<StdMutex<VecDeque<Value>>>>, Json(body): Json<Value>) -> Json<Value> {
+        let mut seen = seen.lock().unwrap();
+        seen.push_back(body);
+        let n = seen.len();
+        Json(json!({ "access_token": format!("a{n}"), "refresh_token": format!("r{}", n + 1), "expires_in": 3600 }))
+    }
+    let app = Router::new()
+        .route("/api/device/token", post(token))
+        .with_state(refreshes.clone());
+    let http = TcpListener::bind("127.0.0.1:0").await.unwrap();
+    let http_addr = http.local_addr().unwrap();
+    tokio::spawn(async move { axum::serve(http, app).await.unwrap() });
+
+    // The cloud serves both on one port: a POST goes on to the token endpoint, anything else is a socket.
+    let (tx, rx) = mpsc::unbounded_channel();
+    let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
+    let addr = listener.local_addr().unwrap();
+    tokio::spawn(async move {
+        loop {
+            let (stream, _) = listener.accept().await.unwrap();
+            let mut head = [0u8; 4];
+            if stream.peek(&mut head).await.unwrap_or(0) == 4 && &head == b"POST" {
+                // The token's POST: hand the bytes to the HTTP server.
+                tokio::spawn(async move {
+                    let mut upstream = tokio::net::TcpStream::connect(http_addr).await.unwrap();
+                    let mut stream = stream;
+                    let _ = tokio::io::copy_bidirectional(&mut stream, &mut upstream).await;
+                });
+                continue;
+            }
+            let tx = tx.clone();
+            tokio::spawn(async move {
+                let seen = Arc::new(StdMutex::new((String::new(), String::new())));
+                let kept = seen.clone();
+                let callback = move |req: &Request, mut res: Response| {
+                    let protocols = req
+                        .headers()
+                        .get("Sec-WebSocket-Protocol")
+                        .and_then(|v| v.to_str().ok());
+                    *kept.lock().unwrap() = (req.uri().to_string(), protocols.unwrap_or("").to_string());
+                    res.headers_mut()
+                        .insert("Sec-WebSocket-Protocol", HeaderValue::from_static(SUBPROTOCOL));
+                    Ok(res)
+                };
+                if let Ok(ws) = tokio_tungstenite::accept_hdr_async(stream, callback).await {
+                    let (uri, protocols) = seen.lock().unwrap().clone();
+                    let _ = tx.send(Accepted { uri, protocols, ws });
+                }
+            });
+        }
+    });
+    (format!("http://{addr}"), refreshes, rx)
+}

+/// Where refresh tokens live, keyed by cloud URL (link.ts `TokenStore`).
+pub trait TokenStore: Send + Sync {
+    fn get(&self, url: &str) -> Result<Option<String>>;

+/// The OS credential store's service name for the link tokens (link.ts `KEYCHAIN_SERVICE`).
+const KEYCHAIN_SERVICE: &str = "retaskable";

+/// The file used when there's no OS credential store: `<data dir>/credentials.json`.
+pub fn credentials_file(data_dir: &Path) -> PathBuf {

+/// The link tokens' store (link.ts `keychain`): in the file `RETASKABLE_CREDENTIALS_FILE` names, else in the OS
+/// credential store when it works, else in the data folder's file. The override keeps an agent started to check a
+/// build off this computer's own link (credentials.ts `credentialsFileOverride`).
+pub fn keychain(data_dir: &Path, env: impl Fn(&str) -> Option<String>) -> Arc<dyn TokenStore> {
+    if let Some(file) = env("RETASKABLE_CREDENTIALS_FILE").filter(|f| !f.trim().is_empty()) {
+        return Arc::new(FileStore::new(PathBuf::from(file.trim()), KEYCHAIN_SERVICE));
+    }
+    if os_store_works() {
+        Arc::new(OsStore(KEYCHAIN_SERVICE))
+    } else {
+        Arc::new(FileStore::new(credentials_file(data_dir), KEYCHAIN_SERVICE))
+    }
+}

+/// The line at start on a machine using a file, else None (credentials.ts `credentialFileNote`).
+pub fn credential_file_note(data_dir: &Path, env: impl Fn(&str) -> Option<String>) -> Option<String> {
+    if let Some(file) = env("RETASKABLE_CREDENTIALS_FILE").filter(|f| !f.trim().is_empty()) {
+        return Some(format!(
+            "Link tokens and keys are kept in {} (RETASKABLE_CREDENTIALS_FILE), not the OS credential store.",
+            file.trim()
+        ));
+    }
+    (!os_store_works()).then(|| {

+            "No OS credential store here: link tokens and keys are kept in {}, readable by this user only.",

+/// Whether this machine's OS credential store answers: a read of an entry nobody writes. Always yes on Windows and
+/// macOS. On Linux, in a Docker container there's no Secret Service, so the read fails.
+fn os_store_works() -> bool {
+    if !cfg!(target_os = "linux") {
+        return true;
+    }
+    match keyring::Entry::new("retaskable-probe", "probe").and_then(|e| e.get_password()) {
+        Ok(_) | Err(keyring::Error::NoEntry) => true,
+        Err(_) => false,
+    }
+}

+/// The OS credential store, one entry per URL under the service. Its Windows target is `<url>.<service>`, the same as
+/// the TypeScript agent's, so either agent finds the other's link.
+struct OsStore(&'static str);
+
+impl TokenStore for OsStore {
+    fn get(&self, url: &str) -> Result<Option<String>> {
+        match keyring::Entry::new(self.0, url).and_then(|e| e.get_password()) {
+            Ok(token) => Ok(Some(token).filter(|t| !t.is_empty())),
+            Err(keyring::Error::NoEntry) => Ok(None),
+            Err(e) => Err(e).context("reading the OS credential store"),
+        }
+    }
+    fn set(&self, url: &str, token: &str) -> Result<()> {
+        keyring::Entry::new(self.0, url)
+            .and_then(|e| e.set_password(token))
+            .context("writing to the OS credential store")
+    }
+    fn delete(&self, url: &str) -> Result<()> {
+        match keyring::Entry::new(self.0, url).and_then(|e| e.delete_credential()) {
+            Ok(()) | Err(keyring::Error::NoEntry) => Ok(()),
+            Err(e) => Err(e).context("deleting from the OS credential store"),
+        }
+    }
+}

+type Saved = BTreeMap<String, BTreeMap<String, String>>;

+/// Secrets in one JSON file, keyed by service then account (credentials.ts `fileEntry`). Read on every call, so two
+/// processes see each other's writes.
+pub struct FileStore {
+    file: PathBuf,
+    service: String,
+}
+
+impl FileStore {
+    pub fn new(file: PathBuf, service: &str) -> Self {
+        Self {
+            file,
+            service: service.to_string(),
+        }
+    }
+
+    fn read(&self) -> Result<Saved> {
+        match std::fs::read_to_string(&self.file) {
+            Ok(text) => serde_json::from_str(&text).with_context(|| format!("Couldn't read {}", self.file.display())),
+            Err(e) if e.kind() == ErrorKind::NotFound => Ok(Saved::new()),
+            Err(e) => Err(e).with_context(|| format!("Couldn't read {}", self.file.display())),
+        }
+    }
+
+    /// Written whole to a temp file readable by this user only, then renamed over, so it's never readable by others
+    /// nor half written.
+    fn write(&self, saved: &Saved) -> Result<()> {
+        if let Some(dir) = self.file.parent().filter(|d| !d.as_os_str().is_empty()) {
+            std::fs::create_dir_all(dir).with_context(|| format!("creating {}", dir.display()))?;
+        }
+        let name = self
+            .file
+            .file_name()
+            .context("the credentials file has no name")?
+            .to_string_lossy();
+        let tmp = self.file.with_file_name(format!("{name}.{}.tmp", std::process::id()));
+        let mut options = std::fs::OpenOptions::new();
+        options.write(true).create(true).truncate(true);
+        #[cfg(unix)]
+        std::os::unix::fs::OpenOptionsExt::mode(&mut options, 0o600);
+        let file = options
+            .open(&tmp)
+            .with_context(|| format!("writing {}", tmp.display()))?;
+        #[cfg(unix)]
+        std::fs::set_permissions(&tmp, std::os::unix::fs::PermissionsExt::from_mode(0o600))?;
+        serde_json::to_writer_pretty(file, saved).with_context(|| format!("writing {}", tmp.display()))?;
+        std::fs::rename(&tmp, &self.file).with_context(|| format!("writing {}", self.file.display()))
+    }
+}
+
+impl TokenStore for FileStore {
+    fn get(&self, url: &str) -> Result<Option<String>> {
+        Ok(self
+            .read()?
+            .get(&self.service)
+            .and_then(|s| s.get(url))
+            .filter(|t| !t.is_empty())
+            .cloned())
+    }
+    fn set(&self, url: &str, token: &str) -> Result<()> {
+        let mut saved = self.read()?;
+        saved
+            .entry(self.service.clone())
+            .or_default()
+            .insert(url.to_string(), token.to_string());
+        self.write(&saved)
+    }
+    fn delete(&self, url: &str) -> Result<()> {
+        let mut saved = self.read()?;
+        let Some(service) = saved.get_mut(&self.service) else {
+            return Ok(());
+        };
+        if service.remove(url).is_none() {
+            return Ok(());
+        }
+        if service.is_empty() {
+            saved.remove(&self.service);
+        }
+        self.write(&saved)
+    }
+}

+/// The code to show while a link waits for approval.
+#[derive(Debug, Clone, PartialEq)]
+pub struct DeviceCode {
+    pub user_code: String,
+    pub verification_uri: String,
+    pub verification_uri_complete: String,
+    pub expires_in: u64,
+}

+/// One link's device flow (link.ts `link`).
+pub struct Link<'a> {
+    pub http: reqwest::Client,
+    pub store: &'a dyn TokenStore,
+    /// `machine_info`, sent with the code request.
+    pub machine: Value,
+    pub target: LinkTarget,
+    /// What one second of the server's polling interval is: a second, or less in tests.
+    pub second: Duration,
+}
+
+impl Link<'_> {
+    /// Asks `url` for a device code, shows it with `on_code`, polls until approved, and stores the refresh token.
+    /// The agent's id on the cloud when it's done.
+    pub async fn run(&self, url: &str, on_code: impl FnOnce(&DeviceCode)) -> Result<u64> {
+        let path = self.target.device_path;
+        let (ok, code) = self.post(url, &format!("{path}/code"), &self.machine).await?;
+        if !ok {
+            bail!("Couldn't get a code: {}", describe(&code));
+        }
+        let device_code = text(&code, "device_code");
+        let mut interval = code
+            .get("interval")
+            .and_then(Value::as_u64)
+            .filter(|i| *i > 0)
+            .unwrap_or(5);
+        on_code(&DeviceCode {
+            user_code: text(&code, "user_code"),
+            verification_uri: text(&code, "verification_uri"),
+            verification_uri_complete: text(&code, "verification_uri_complete"),
+            expires_in: code
+                .get("expires_in")
+                .and_then(Value::as_u64)
+                .filter(|e| *e > 0)
+                .unwrap_or(600),
+        });
+        loop {
+            tokio::time::sleep(self.second * interval as u32).await;
+            let body = json!({ "grant_type": DEVICE_CODE_GRANT, "device_code": device_code });
+            let (ok, res) = self.post(url, &format!("{path}/token"), &body).await?;
+            if ok {
+                let refresh = res
+                    .get("refresh_token")
+                    .and_then(Value::as_str)
+                    .filter(|t| !t.is_empty());
+                let Some(refresh) = refresh else {
+                    bail!("The {} approved the link but sent no refresh token.", self.target.name);
+                };
+                self.store.set(url, refresh)?;
+                return Ok(res.get("agent_id").and_then(Value::as_u64).unwrap_or(0));
+            }
+            match res.get("error").and_then(Value::as_str) {
+                Some("authorization_pending") => {}
+                Some("slow_down") => interval += 5, // RFC 8628 section 3.5
+                Some("access_denied") => bail!("The link was denied on the /link page."),
+                Some("expired_token") => bail!("The code expired before it was approved. Run the link again."),
+                _ => bail!("Linking failed: {}", describe(&res)),
+            }
+        }
+    }
+
+    /// POSTs JSON; whether it answered 2xx, and its JSON body.
+    async fn post(&self, url: &str, path: &str, body: &Value) -> Result<(bool, Map<String, Value>)> {
+        let res = self
+            .http
+            .post(format!("{url}{path}"))
+            .json(body)
+            .send()
+            .await
+            .map_err(|e| anyhow!("Couldn't reach {url}: {}. Check {}.", cause(&e), self.target.setting))?;
+        let status = res.status();
+        match res.json::<Value>().await {
+            Ok(Value::Object(json)) => Ok((status.is_success(), json)),
+            _ => bail!(
+                "{url}{path} answered {} without JSON. Is {url} a Retaskable {}?",
+                status.as_u16(),
+                self.target.name
+            ),
+        }
+    }
+}
+
+fn text(json: &Map<String, Value>, key: &str) -> String {
+    match json.get(key) {
+        Some(Value::String(s)) => s.clone(),
+        Some(v) => v.to_string(),
+        None => String::new(),
+    }
+}
+
+fn describe(json: &Map<String, Value>) -> String {
+    ["error_description", "error", "message"]
+        .iter()
+        .find_map(|k| {
+            json.get(*k)
+                .map(|v| v.as_str().map(String::from).unwrap_or_else(|| v.to_string()))
+        })
+        .unwrap_or_else(|| Value::Object(json.clone()).to_string())
+}
+
+/// The innermost cause of a request's failure ("connection refused"), as the TypeScript agent's `causeMessage`.
+pub(crate) fn cause(err: &(dyn std::error::Error + 'static)) -> String {
+    let mut inner = err;
+    while let Some(next) = inner.source() {
+        inner = next;
+    }
+    inner.to_string()
+}

+/// An access token for one cloud, or why there's none.
+#[derive(Debug, Clone, PartialEq)]
+pub enum TokenAnswer {
+    Token {
+        url: String,
+        access_token: String,
+    },
+    /// Not linked: look again later.
+    Idle,
+    /// The cloud couldn't be reached.
+    Unreachable,
+}
+
+/// An access token is used until a minute before it expires.
+const TOKEN_MARGIN: Duration = Duration::from_secs(60);
+
+struct Held {
+    access_token: String,
+    until: Instant,
+}
+
+/// One link's access token, refreshed from the stored refresh token when it runs out (link.ts `DeviceTokens`).
+/// Refresh tokens rotate on every use and reusing one unlinks the agent, so there's only ever one refresh under way:
+/// the lock is held across it.
+pub struct DeviceTokens {
+    url: String,
+    target: LinkTarget,
+    store: Arc<dyn TokenStore>,
+    http: reqwest::Client,
+    held: Mutex<Option<Held>>,
+}
+
+impl DeviceTokens {
+    pub fn new(url: String, target: LinkTarget, store: Arc<dyn TokenStore>, http: reqwest::Client) -> Self {
+        Self {
+            url,
+            target,
+            store,
+            http,
+            held: Mutex::new(None),
+        }
+    }
+
+    /// The token in hand while it has a minute left, else a fresh one. `stale`: a token the caller found no good,
+    /// which isn't handed out again.
+    pub async fn access_token(&self, stale: Option<&str>) -> TokenAnswer {
+        let mut held = self.held.lock().await;
+        if let Some(h) = held.as_ref()
+            && Some(h.access_token.as_str()) != stale
+            && Instant::now() < h.until
+        {
+            return TokenAnswer::Token {
+                url: self.url.clone(),
+                access_token: h.access_token.clone(),
+            };
+        }
+        *held = None;
+        let (answer, fresh) = self.refresh().await;
+        *held = fresh;
+        answer
+    }
+
+    /// Waits up to `wait` for a refresh under way to answer and store its new refresh token. Exiting in the middle of
+    /// one loses the token the server just rotated to, and the next start's replay of the old one unlinks the agent.
+    /// `then` runs with no refresh able to start, to stop whatever would start one.
+    pub async fn settled(&self, wait: Duration, then: impl FnOnce()) {
+        let _held = tokio::time::timeout(wait, self.held.lock()).await;
+        then();
+    }
+
+    /// Swaps the refresh token for an access token, storing the new refresh token at once.
+    async fn refresh(&self) -> (TokenAnswer, Option<Held>) {
+        let url = &self.url;
+        let refresh_token = match self.store.get(url) {
+            Ok(Some(token)) => token,
+            Ok(None) => return (TokenAnswer::Idle, None), // not linked (yet)
+            Err(e) => {
+                tracing::warn!("cloud ({url}): not connecting: {e:#}");
+                return (TokenAnswer::Idle, None);
+            }
+        };
+        let body = json!({ "grant_type": "refresh_token", "refresh_token": refresh_token });
+        let res = match self
+            .http
+            .post(format!("{url}{}/token", self.target.device_path))
+            .json(&body)
+            .send()
+            .await
+        {
+            Ok(res) => res,
+            Err(e) => {
+                tracing::warn!("cloud ({url}): couldn't reach {url}: {}", cause(&e));
+                return (TokenAnswer::Unreachable, None);
+            }
+        };
+        let status = res.status();
+        let json = res.json::<Value>().await.unwrap_or(Value::Null);
+        let access = json.get("access_token").and_then(Value::as_str);
+        let rotated = json.get("refresh_token").and_then(Value::as_str);
+        if let (true, Some(access), Some(rotated)) = (status.is_success(), access, rotated) {
+            if let Err(e) = self.store.set(url, rotated) {
+                tracing::warn!("cloud ({url}): couldn't keep the new refresh token: {e:#}");
+            }
+            let seconds = json.get("expires_in").and_then(Value::as_u64).unwrap_or(0);
+            let expires = Instant::now() + Duration::from_secs(seconds);
+            let until = expires.checked_sub(TOKEN_MARGIN).unwrap_or(expires);
+            let answer = TokenAnswer::Token {
+                url: url.clone(),
+                access_token: access.to_string(),
+            };
+            return (
+                answer,
+                Some(Held {
+                    access_token: access.to_string(),
+                    until,
+                }),
+            );
+        }
+        let said = |k: &str| json.get(k).and_then(Value::as_str);
+        if said("error") == Some("invalid_grant") {
+            // Unlinked on the website (or the token was reused): the stored token is no good any more.
+            let why = said("error_description").unwrap_or("invalid_grant");
+            tracing::warn!("cloud ({url}): not linked any more: {why}");
+            if let Err(e) = self.store.delete(url) {
+                tracing::warn!("cloud ({url}): couldn't delete the old token: {e:#}");
+            }
+            return (TokenAnswer::Idle, None);
+        }
+        let why = said("error_description").or_else(|| said("error")).unwrap_or("");
+        tracing::warn!(
+            "cloud ({url}): refreshing the access token failed: {} {why}",
+            status.as_u16()
+        );
+        (TokenAnswer::Unreachable, None)
+    }
+}

+/// `retaskable link [production|staging|local]`: the device flow in the terminal. The exit code.
+async fn link_command(
+    args: &[String],
+    cfg: &config::ConfigFile,
+    dir: &Path,
+    env: impl Fn(&str) -> Option<String> + Copy,
+) -> i32 {
+    let url = match args {
+        [] => link::cloud_url(cfg, env).map_err(|e| format!("{e:#}")),
+        [name] => link::environment_url(name).map(String::from).ok_or_else(String::new),
+        _ => Err(String::new()),
+    };
+    let url = match url {
+        Ok(url) => url,
+        Err(e) => {
+            if !e.is_empty() {
+                eprintln!("{e}");
+            }
+            eprintln!("Usage: retaskable link [production|staging|local]");
+            return 2;
+        }
+    };
+    let keychain = link::keychain(dir, env);
+    if matches!(keychain.get(&url), Ok(Some(_))) {
+        println!("Already linked to {url}. Linking again replaces that link.");
+    }
+    println!("Linking to {url}…");
+    let flow = link::Link {
+        http: reqwest::Client::new(),
+        store: keychain.as_ref(),
+        machine: link::machine_info(env),
+        target: link::CLOUD_TARGET,
+        second: Duration::from_secs(1),
+    };
+    let shown = |code: &link::DeviceCode| {
+        println!("Go to {} and enter {}", code.verification_uri, code.user_code);
+        if !code.verification_uri_complete.is_empty() {
+            println!("(or open {})", code.verification_uri_complete);
+        }
+        println!(
+            "The code works for {} minutes. Waiting for approval…",
+            code.expires_in.div_ceil(60)
+        );
+    };
+    match flow.run(&url, shown).await {
+        Ok(id) => {
+            println!("Linked to {url} as agent {id}. The refresh token is in this machine's credential store.");
+            0
+        }
+        Err(e) => {
+            eprintln!("{e:#}");
+            1
+        }
+    }
+}
